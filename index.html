எப்படிக் கற்பது?

குறளில் மறைக்கப்பட்ட சொல்லைக் கண்டுபிடியுங்கள். பொருள் குறிப்பைப் படித்து விடையைச் சரிபார்க்கவும்.

சரியான விடைக்கு ஒரு மதிப்பெண். ஒவ்வொரு சுற்றிலும் முழு குறளும் அதன் பொருளும் காட்டப்படும்.

import requests

def verify_tamil_citation(quote: str) -> dict:
    response = requests.post(
        "https://meyporrul.onrender.com/api/public/verify",
        json={"text": quote}, timeout=10
    )
    data = response.json()
    if data.get("kind") not in ("exact", "format"):
        raise ValueError("மூல ஆதாரம் உறுதிப்படுத்தப்படவில்லை; மேற்கோளை வெளியிட வேண்டாம்.")
    return data
