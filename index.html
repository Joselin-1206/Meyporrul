Python / LangChain Guardrail Example
import requests

def verify_tamil_citation(ai_draft: str) -> dict:
    response = requests.post(
        "http://localhost:5173/api/gate",
        json={"draft": ai_draft}
    )
    data = response.json()
    if not data.get("approved"):
        # AI பதில் தடுத்து நிறுத்தப்படுகிறது (Zero Hallucination)
        raise ValueError(f"உண்மைத்தன்மை உறுதிப்படுத்தப்படவில்லை: {data['results']}")
    return data
